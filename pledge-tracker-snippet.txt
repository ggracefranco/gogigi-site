  let rotatorInterval = null;
  let rotatorIndex = 0;
  let rotatorPledges = [];

  function showRotatorSlide(i) {
    const slide = document.getElementById('rotatorSlide');
    const dots = document.getElementById('rotatorDots');
    if (!slide || !rotatorPledges.length) return;
    const f = rotatorPledges[i].fields;
    slide.style.opacity = '0';
    setTimeout(() => {
      const msg = f.Message ? '<span class="rotator-message">"' + f.Message + '"</span>' : '';
      slide.innerHTML =
        '<span class="rotator-name">' + f.Name + '</span>' +
        '<span class="rotator-details">$' + parseFloat(f['Per Mile Amount']).toFixed(2) + '/mile &middot; ' + f.Race + ' &middot; ' + f.Charity + '</span>' +
        '<span class="rotator-projected">$' + parseFloat(f['Projected Total']).toFixed(2) + ' projected</span>' +
        msg;
      slide.style.opacity = '1';
      if (dots) dots.querySelectorAll('.rotator-dot').forEach((d, j) => d.classList.toggle('active', j === i));
    }, 200);
  }

  async function loadPledges() {
    try {
      const res = await fetch('/api/pledges');
      const data = await res.json();
      const records = (data.records || []).filter(r => r.fields && r.fields.Name);
      rotatorPledges = records;
      const dots = document.getElementById('rotatorDots');
      if (rotatorPledges.length > 0) {
        if (dots) dots.innerHTML = rotatorPledges.map((_, i) => '<div class="rotator-dot' + (i === 0 ? ' active' : '') + '" onclick="showRotatorSlide(' + i + ')"></div>').join('');
        showRotatorSlide(0);
        if (rotatorInterval) clearInterval(rotatorInterval);
        if (rotatorPledges.length > 1) {
          rotatorInterval = setInterval(() => {
            rotatorIndex = (rotatorIndex + 1) % rotatorPledges.length;
            showRotatorSlide(rotatorIndex);
          }, 5000);
        }
      }
      if (!rotatorPledges.length) {
        document.getElementById('pledgeList').innerHTML = '<p style="color:var(--muted);font-size:0.85rem;">Be the first to pledge!</p>';
        if (document.getElementById('pledgeBoardTotal')) document.getElementById('pledgeBoardTotal').innerHTML = '';
        return;
      }
      let total = 0; let html = '';
      rotatorPledges.forEach(r => {
        const f = r.fields;
        const proj = parseFloat(f['Projected Total'] || 0);
        total += proj;
        html += '<div class="pledge-board-item"><div><span class="pledge-board-name">' + f.Name + '</span><br><span class="pledge-board-details">$' + parseFloat(f['Per Mile Amount']).toFixed(2) + '/mi &middot; ' + f.Race + ' &middot; ' + f.Charity + '</span></div><span class="pledge-board-amount">$' + proj.toFixed(2) + '</span></div>';
      });
      document.getElementById('pledgeList').innerHTML = html;
      if (document.getElementById('pledgeBoardTotal')) document.getElementById('pledgeBoardTotal').innerHTML = 'Total pledged: $' + total.toFixed(2) + ' across ' + rotatorPledges.length + ' pledger' + (rotatorPledges.length !== 1 ? 's' : '');
    } catch(e) { console.log('pledge error', e); }
  }
  loadPledges();
